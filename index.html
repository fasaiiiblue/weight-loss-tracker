/** * Shape me up Record - Ultra-Fast Engine (Optimized & Clean) * ศูนย์ดูแลและควบคุมน้ำหนัก รพ.สมิติเวชศรีราชา */ function doGet(e) { try { const action = e && e.parameter ? e.parameter.action : ""; const lineUid = e && e.parameter ? e.parameter.lineUid : ""; if (action === "getPatientInfo") { return createJsonResponse(getFastPatientData(lineUid)); } if (action === "getInjectionHistory") { return createJsonResponse(getFastInjectionHistory(lineUid)); } if (action === "ping") { return createJsonResponse({ status: "ok", message: "Connected successfully" }); } return createJsonResponse({ status: "invalid_action" }); } catch (err) { return createJsonResponse({ status: "error", message: err.toString() }); } } function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(5000); const data = JSON.parse(e.postData.contents); const action = data.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); ensureSheetsExist(); // 1. ลงทะเบียนคนไข้ใหม่ (รับเฉพาะข้อมูลจำเป็น) if (action === "register") { const pSheet = ss.getSheetByName("Patients"); const lastRow = pSheet.getLastRow(); let targetRow = -1; if (lastRow > 1) { const uids = pSheet.getRange(2, 1, lastRow - 1, 1).getValues(); for (let i = 0; i < uids.length; i++) { if (String(uids[i][0]).trim() === String(data.lineUid).trim()) { targetRow = i + 2; break; } } } if (targetRow !== -1) { pSheet.getRange(targetRow, 2, 1, 7).setValues([[ data.hn || "-", data.name, "'" + data.phone, data.medication, "-", "สัปดาห์ละครั้ง", data.injectionDay ]]); } else { pSheet.appendRow([ data.lineUid, data.hn || "-", data.name, "'" + data.phone, data.medication, "-", "สัปดาห์ละครั้ง", data.injectionDay, "", "ลงทะเบียนเมื่อ " + Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm"), 1, // last_pen_num 0, // last_dose_num "", // last_injected_date "", // last_site "", // pen_strength 7 // interval_days ]); } return createJsonResponse({ status: "success" }); } // 2. บันทึกการฉีดยาจริง (ตัด LINE Push ออกเพื่อความเร็วระดับเสี้ยววินาที) if (action === "recordInjection") { const pSheet = ss.getSheetByName("Patients"); const logSheet = ss.getSheetByName("Injections_Log"); // กรองคำภาษาอังกฤษและข้อความแปลกๆ ออกจากชื่อตำแหน่งฉีด const cleanSiteName = sanitizeText(data.site || "หน้าท้อง"); const injectDateParts = data.actualDate.split("-"); const injectDate = new Date(Number(injectDateParts[0]), Number(injectDateParts[1]) - 1, Number(injectDateParts[2])); let patientRow = -1; let patientName = data.name || ""; let patientHn = data.hn || "-"; let currentMed = data.medication || ""; const intervalDays = parseInt(data.intervalDays, 10) || 7; const pLastRow = pSheet.getLastRow(); if (pLastRow > 1) { const lastCol = Math.max(pSheet.getLastColumn(), 16); const pData = pSheet.getRange(2, 1, pLastRow - 1, lastCol).getValues(); for (let k = 0; k < pData.length; k++) { if (String(pData[k][0]).trim() === String(data.lineUid).trim()) { patientRow = k + 2; patientHn = pData[k][1] || patientHn; patientName = pData[k][2] || patientName; currentMed = pData[k][4] || currentMed; break; } } } // คำนวณวันนัดถัดไปตามจำนวนวันจริง const nextDue = new Date(injectDate); nextDue.setDate(nextDue.getDate() + intervalDays); const formattedActual = Utilities.formatDate(injectDate, "GMT+7", "dd/MM/yyyy"); const formattedNext = Utilities.formatDate(nextDue, "GMT+7", "dd/MM/yyyy"); const penNum = data.penNumber || 1; const doseNum = data.doseNumber || 1; const cleanDose = formatDoseDisplay(data.currentDoseValue || data.doseDisplay); const cleanPen = formatDoseDisplay(data.penStrength || cleanDose); const clicksText = calculateClicksText(currentMed, cleanDose, cleanPen); const scheduleRemark = `(รอบ ${intervalDays} วัน)`; const logNote = `วันฉีดจริง: ${formattedActual} | นัดรอบถัดไป: ${formattedNext} ${scheduleRemark} | ด้ามที่: ${penNum} | ครั้งที่: ${doseNum} | คำแนะนำคลิก: ${clicksText}`; logSheet.appendRow([ Utilities.getUuid(), new Date(), data.lineUid, patientHn, patientName, currentMed, data.doseDisplay || cleanDose, cleanSiteName, sanitizeText(data.sideEffects || "ปกติ"), logNote ]); if (patientRow !== -1) { pSheet.getRange(patientRow, 5, 1, 3).setValues([[currentMed, cleanDose, `ทุก ${intervalDays} วัน`]]); pSheet.getRange(patientRow, 11, 1, 6).setValues([[ penNum, doseNum, data.actualDate, cleanSiteName, cleanPen, intervalDays ]]); } return createJsonResponse({ status: "success", doseFormatted: cleanDose, clicksInfo: clicksText, nextDueDate: `${formattedNext} ${scheduleRemark}` }); } } catch (err) { return createJsonResponse({ status: "error", message: err.toString() }); } finally { lock.releaseLock(); } } function createJsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function ensureSheetsExist() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let pSheet = ss.getSheetByName("Patients"); if (!pSheet) { pSheet = ss.insertSheet("Patients"); pSheet.appendRow([ "Line UID", "HN", "ชื่อ-สกุล", "เบอร์โทร", "ตัวยา", "ขนาดยา", "ความถี่", "วันนัดฉีด", "แพทย์ผู้ดูแล", "หมายเหตุ", "last_pen_num", "last_dose_num", "last_injected_date", "last_site", "pen_strength", "interval_days" ]); } let lSheet = ss.getSheetByName("Injections_Log"); if (!lSheet) { lSheet = ss.insertSheet("Injections_Log"); lSheet.appendRow([ "Log ID", "Timestamp", "Line UID", "HN", "ชื่อ-สกุล", "ตัวยา", "ขนาดยา", "ตำแหน่งฉีด", "อาการข้างเคียง", "หมายเหตุ" ]); } } function getFastPatientData(lineUid) { if (!lineUid) return { found: false }; const ss = SpreadsheetApp.getActiveSpreadsheet(); const pSheet = ss.getSheetByName("Patients"); if (!pSheet) return { found: false }; const pData = pSheet.getDataRange().getValues(); if (pData.length < 2) return { found: false }; const uidTarget = String(lineUid).trim(); for (let i = 1; i < pData.length; i++) { const row = pData[i]; if (String(row[0]).trim() === uidTarget) { const med = String(row[4] || ""); const currentDose = formatDoseDisplay(row[5]); const penStrength = row[14] ? formatDoseDisplay(row[14]) : currentDose; const intervalDays = row[15] ? parseInt(row[15], 10) : 7; const clicksText = calculateClicksText(med, currentDose, penStrength); let lastDateIso = ""; if (row[12]) { if (row[12] instanceof Date) { lastDateIso = Utilities.formatDate(row[12], "GMT+7", "yyyy-MM-dd"); } else { lastDateIso = String(row[12]).trim(); } } return { found: true, hn: String(row[1] || ""), name: String(row[2] || ""), phone: String(row[3] || ""), medication: med, currentDose: currentDose, penStrength: penStrength, intervalDays: intervalDays, clicksGuidance: clicksText, frequency: String(row[6] || "สัปดาห์ละครั้ง"), injectionDay: String(row[7] || "วันจันทร์"), doctor: String(row[8] || ""), lastPenNumber: row[10] ? parseInt(row[10], 10) : 1, lastDoseNumber: row[11] ? parseInt(row[11], 10) : 0, lastInjectedDate: lastDateIso, lastInjectedSite: sanitizeText(String(row[13] || "")) }; } } return { found: false }; } function getFastInjectionHistory(lineUid) { if (!lineUid) return []; const ss = SpreadsheetApp.getActiveSpreadsheet(); const lSheet = ss.getSheetByName("Injections_Log"); if (!lSheet) return []; const data = lSheet.getDataRange().getValues(); if (data.length < 2) return []; const list = []; const uidTarget = String(lineUid).trim(); // ดึง 30 รายการล่าสุดจากล่างขึ้นบน for (let i = data.length - 1; i >= 1; i--) { const row = data[i]; if (String(row[2]).trim() === uidTarget) { const note = String(row[9] || ""); let actualDate = ""; if (row[1] instanceof Date) { actualDate = Utilities.formatDate(row[1], "GMT+7", "dd/MM/yyyy"); } const mDate = note.match(/วันฉีดจริง:\s*(\d{2}\/\d{2}\/\d{4})/); if (mDate) actualDate = mDate[1]; let penNum = 1; const mPen = note.match(/ด้ามที่:\s*(\d+)/); if (mPen) penNum = parseInt(mPen[1], 10); let doseNum = 1; const mDose = note.match(/ครั้งที่:\s*(\d+)/); if (mDose) doseNum = parseInt(mDose[1], 10); list.push({ penNumber: penNum, doseNumber: doseNum, actualDate: actualDate, medication: String(row[5] || ""), dose: formatDoseDisplay(row[6]), site: sanitizeText(String(row[7] || "")), sideEffects: sanitizeText(String(row[8] || "ปกติ")) }); if (list.length >= 30) break; } } return list; } // กรองคำแปลกๆ เช่น \vert{}, Xert และตัวย่อภาษาอังกฤษ RUQ function sanitizeText(text) { if (!text) return ""; let s = String(text); s = s.replace(/\s*\([A-Z]+\)/gi, ""); // ลบ (RUQ), (LUQ) s = s.replace(/\\?x?v?ert\{\}/gi, ""); // ลบ \vert{}, vert{}, ert{}, Xert s = s.replace(/xert/gi, ""); s = s.replace(/[\\|{}]/g, ""); // ลบเครื่องหมายตกค้าง return s.trim(); } function formatDoseDisplay(val) { if (!val || val === "-") return ""; const m = String(val).match(/([0-9.]+)/); if (!m) return String(val); const num = parseFloat(m[1]); if (isNaN(num) || num <= 0) return String(val); const numStr = (num % 1 === 0) ? num.toFixed(0) : num.toString(); return numStr + " mg"; } function calculateClicksText(med, doseVal, penVal) { const parseNum = (v) => { const m = String(v).match(/([0-9.]+)/); return m ? parseFloat(m[1]) : 0; }; const dose = parseNum(doseVal); const pen = parseNum(penVal); if (!dose || !pen) return ""; if (Math.abs(dose - pen) < 0.01) { return "หมุนถึงเลข 1 (เต็มโดส)"; } let clicks = 0; if (String(med).indexOf("Mounjaro") !== -1 || String(med).indexOf("มอนจาโร") !== -1) { clicks = Math.round((dose / pen) * 60); } else { const baseClicks = (pen <= 0.5) ? 32 : 72; clicks = Math.round((dose / pen) * baseClicks); } if (clicks <= 0) return ""; return "หมุน " + clicks + " คลิก"; }
